\chapter{Definitionen}
\michael{Schreib am Kapitelanfang immer ein zwei Einführungssätze, was hier passiert. Du kannst auch zwischen den Definitionen ein paar Überleitungen schreiben, damit sich das nicht liest wie ein Skript.}
\begin{definition}[Überschneidung]
    Zwei Kanten gelten als überschneidend, wenn sie sich in der Ebene kreuzen.
    Im Folgenden bezeichnet die Funktion \(\intersection: (\mathbb{Q}^2 \times \mathbb{Q}^2) \to \mathcal{P}(\mathbb{Q}^2 \times \mathbb{Q}^2)\) die Menge aller Kanten, die eine gegebene Kante schneiden.
\end{definition}

\begin{definition}[Überschneidung Dreiecke]
    Zwei Dreiecke gelten als überschneidend, wenn sie mindestens einen inneren Punkt gemeinsam haben.
    Das bedeutet, dass sich zwei Außenkanten kreuzen und sich nicht nur berühren dürfen.
    Im Folgenden bezeichnet die Funktion \(\intersection: (\mathbb{Q}^2 \times \mathbb{Q}^2 \times \mathbb{Q}^2) \to \mathcal{P}(\mathbb{Q}^2 \times \mathbb{Q}^2 \times \mathbb{Q}^2)\) die Menge aller Dreiecke, die ein gegebenes Dreieck schneiden.
\end{definition}

\michael{Triangulierung ist überlicher als Triangulation..}
\begin{definition}[Triangulation]
    Eine Triangulation einer Punktemenge \(P \subset \mathbb{Q}^2\) ist ein maximaler kreuzungsfreier Graph auf \(P\).
    Das bedeutet, es können keine weiteren Kanten hinzugefügt werden, ohne dass sich Kanten schneiden.
\end{definition}

\begin{definition}[Grad]
    Der Grad eines Knotens ist die Anzahl der anliegenden Kanten.
    Im Folgenden wird er durch die Funktion \(\degree: \mathbb{Q}^2 \to \mathbb{N}\) angegeben.
\end{definition}

\begin{definition}[Gradbeschränkte Triangulation]
    Die Gradbeschränkte Triangulation hat die gleichen Vorraussetzungen wie eine Triangulation.
    Zusätzlich gibt es noch eine Funktion \(\degrees: \mathbb{Q}^2 \longrightarrow \mathbb{Q}\).
    Diese gibt den Grad an welcher ein Knoten haben muss.
\end{definition}

\michael{Du kannst bei sowas auch ruhig eine Figure machen mit Triangulierung + Flip Highlight.}
\begin{definition}[Flip]
    \label{def:flip}
    Ein Flip ist eine Operation, bei der eine Kante entfernt und eine andere hinzugefügt wird, sodass die Eigenschaften einer Triangulation erhalten bleiben.
    In dieser Arbeit wird nur der Diagonale Flip betrachtet, bei dem die Diagonale eines konvexen Vierecks durch die andere Diagonale ersetzt wird.
    Ein Flip wird durch die Funktion \(\flip: (\mathbb{Q}^2 \times \mathbb{Q}^2) \to (\mathbb{Q}^2 \times \mathbb{Q}^2)\) definiert.
\end{definition}

\begin{definition}[Konvexe Hülle mit Randpunkten]
    Die konvexe Hülle einer Punktmenge ist die kleinste konvexe Menge, die alle Punkte enthält.
    Die Randpunkte sind die Punkte, die auf dem Rand der konvexen Hülle liegen, einschließlich solcher, die nicht an einer Ecke liegen.
    Die konvexe Hülle inklusive Randpunkte wird durch die Funktion \(\convexhull: \mathcal{P}(\mathbb{Q}^2) \to \mathcal{P}(\mathbb{Q}^2)\) definiert.
\end{definition}

\begin{definition}[Konjunktive Normalform]
    Eine Formel in konjunktiver Normalform (KNF) ist eine logische Formel, die als Konjunktion (UND-Verknüpfung) von Klauseln dargestellt wird, wobei jede Klausel eine Disjunktion (ODER-Verknüpfung) von Literalen ist.
    Ein Literal ist entweder eine Variable oder ihre Negation.
    Ein Beispiel für eine KNF ist \[(x_1 \lor \overline{x_2}) \land (x_2 \lor x_3)\] ,wobei \(x_1\), \(x_2\) und \(x_3\) Variablen sind. Eine Negation wird durch das Symbol \(\overline{x}\) dargestellt.
\end{definition}
