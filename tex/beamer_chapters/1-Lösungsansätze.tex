    \item Beispiel v1,v2,v3
    \item wenn v2 grad 2
    \begin{itemize}
        \item e1 aktiv
        \item sonnst muss Kante wie e2 aktiv
        \item sollgrad von v2 zu hoch
    \end{itemize}
    \item wenn v2 größer 2
    \begin{itemize}
        \item e1 nicht aktiv
        \item sollgrad von v2 sonnst nicht erfüllbar
    \end{itemize}
    \item v2 kleiner 2 nicht möglich
    \begin{itemize}
        \item in Triangulierung
    \end{itemize}
}
